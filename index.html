// otherwise this folder is found or created in My Drive var SNAP_KEEP_DAYS = 120; // older daily files are moved to trash var SNAP_PART_CHARS = 8000000; // base64 characters per download part /* ---------------------------------------------------------------- helpers */ function ss_() { return SpreadsheetApp.openById(SPREADSHEET_ID); } function baseSheet_() { var ss = ss_(); if (BASE_SHEET_NAME) { return ss.getSheetByName(BASE_SHEET_NAME) || ss.insertSheet(BASE_SHEET_NAME); } return ss.getSheets()[0]; } function namedSheet_(name, header) { var ss = ss_(), sh = ss.getSheetByName(name); } function metaSheet_() { return namedSheet_(META_SHEET_NAME, ['Uploaded_At', 'File_Name', 'Source_Sheet', 'Rows', 'Columns', 'Upload_Id', 'By']); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function ok_(obj) { obj = obj || {}; obj.ok = true; return json_(obj); } function err_(msg) { return json_({ ok: false, error: String(msg) }); } function body_(e) { if (!e || !e.postData || !e.postData.contents) return {}; try { return JSON.parse(e.postData.contents); } catch (x) { return {}; } } function who_() { try { return Session.getActiveUser().getEmail() || 'anonymous'; } catch (x) { return 'anonymous'; } } /* ------------------------------------------------------------------- read */ function doGet(e) { var p = (e && e.parameter) || {}; var action = p.action || 'ping'; try { if (action === 'ping') return ok_({ version: API_VERSION, sheet: baseSheet_().getName() }); if (action === 'meta') return ok_(readMeta_()); if (action === 'rows') return ok_(readRows_(Number(p.offset || 0), Number(p.limit || 5000))); if (action === 'rca') return ok_({ notes: readRca_() }); if (action === 'snaps') return ok_({ snaps: listSnaps_() }); if (action === 'snap') return ok_(readSnap_(p.day || '', Number(p.part || 0))); return err_('Unknown action: ' + action); } catch (x) { return err_(x.message); } } function readMeta_() { var sh = baseSheet_(); var rows = Math.max(0, sh.getLastRow() - 1), cols = sh.getLastColumn(); var ms = metaSheet_(), last = ms.getLastRow(), info = {}; if (last > 1) { var v = ms.getRange(last, 1, 1, 7).getValues()[0]; info = { uploadedAt: v[0] ? new Date(v[0]).getTime() : null, fileName: v[1], sourceSheet: v[2], uploadId: v[5], by: v[6] }; } info.rows = rows; info.cols = cols; info.sheet = sh.getName(); return info; } function readRows_(offset, limit) { var sh = baseSheet_(); var lastRow = sh.getLastRow(), lastCol = sh.getLastColumn(); if (lastRow < 2 || lastCol < 1) return { headers: [], rows: [], total: 0, offset: 0, done: true }; var total = lastRow - 1; limit = Math.min(Math.max(limit, 1), 10000); var out = { total: total, offset: offset }; if (offset === 0) out.headers = sh.getRange(1, 1, 1, lastCol).getValues()[0]; if (offset >= total) { out.rows = []; out.done = true; return out; } var take = Math.min(limit, total - offset); // Display values keep the sheet's own text for codes and IDs; the dashboard parses numbers itself. out.rows = sh.getRange(2 + offset, 1, take, lastCol).getDisplayValues(); out.done = (offset + take) >= total; return out; } function readRca_() { var sh = rcaSheet_(), last = sh.getLastRow(); if (last < 2) return {}; var vals = sh.getRange(2, 1, last - 1, 4).getValues(), out = {}; for (var i = 0; i < vals.length; i++) { var k = String(vals[i][0] || '').trim(); if (k) out[k] = String(vals[i][3] === null || vals[i][3] === undefined ? '' : vals[i][3]); } return out; } /* ------------------------------------------------------------------ write */ function doPost(e) { var b = body_(e), action = b.action || ''; var lock = LockService.getScriptLock(); try { // An upload is several calls; serialise them so two people cannot interleave chunks. if (!lock.tryLock(30000)) return err_('Another upload is in progress. Try again in a moment.'); if (action === 'begin') return ok_(begin_(b)); if (action === 'chunk') return ok_(chunk_(b)); if (action === 'commit') return ok_(commit_(b)); if (action === 'rca') return ok_(saveRca_(b)); if (action === 'snapPart') return ok_(snapPart_(b)); if (action === 'snapCommit') return ok_(snapCommit_(b)); return err_('Unknown action: ' + action); } catch (x) { return err_(x.message); } finally { try { lock.releaseLock(); } catch (x2) {} } } /** Clear the base tab and write the header row. Each upload replaces what is there. */ function begin_(b) { var headers = b.headers || []; if (!headers.length) throw new Error('begin: no headers supplied'); var sh = baseSheet_(); sh.clear(); sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); sh.setFrozenRows(1); var props = PropertiesService.getScriptProperties(); props.setProperty('uploadId', String(b.uploadId || '')); props.setProperty('uploadCols', String(headers.length)); props.setProperty('uploadWritten', '0'); return { uploadId: b.uploadId, headers: headers.length, sheet: sh.getName() }; } /** Append one chunk of rows in sheet order. */ function chunk_(b) { var rows = b.rows || []; if (!rows.length) return { written: 0 }; var props = PropertiesService.getScriptProperties(); if (b.uploadId && props.getProperty('uploadId') && b.uploadId !== props.getProperty('uploadId')) { throw new Error('This upload was superseded by a newer one. Re-upload to continue.'); } var sh = baseSheet_(); var cols = Number(props.getProperty('uploadCols') || rows[0].length); var norm = rows.map(function (r) { var out = r.slice(0, cols); while (out.length < cols) out.push(''); return out; }); sh.getRange(sh.getLastRow() + 1, 1, norm.length, cols).setValues(norm); var written = Number(props.getProperty('uploadWritten') || 0) + norm.length; props.setProperty('uploadWritten', String(written)); return { written: norm.length, totalWritten: written }; } /** Record the upload in Upload_Meta. */ function commit_(b) { var sh = baseSheet_(), ms = metaSheet_(); var rows = Math.max(0, sh.getLastRow() - 1); ms.appendRow([new Date(), b.fileName || '', b.sourceSheet || '', rows, sh.getLastColumn(), b.uploadId || '', who_()]); // Keep the meta log short. var last = ms.getLastRow(); if (last > 201) ms.deleteRows(2, last - 201); return { rows: rows, cols: sh.getLastColumn(), sheet: sh.getName() }; } /** Upsert one RCA note, or a batch of them. */ function saveRca_(b) { var notes = b.notes || (b.key ? [{ key: b.key, text: b.text }] : []); if (!notes.length) return { saved: 0 }; var sh = rcaSheet_(), last = sh.getLastRow(); var keys = last > 1 ? sh.getRange(2, 1, last - 1, 1).getValues() : []; var rowOf = {}; for (var i = 0; i < keys.length; i++) rowOf[String(keys[i][0]).trim()] = i + 2; var now = new Date(), user = who_(), appended = []; for (var j = 0; j < notes.length; j++) { var n = notes[j], key = String(n.key || '').trim(); if (!key) continue; var parts = key.split('|'); var row = [key, parts[1] || '', parts.slice(2).join('|'), String(n.text || ''), now, user]; if (rowOf[key]) sh.getRange(rowOf[key], 1, 1, 6).setValues([row]); else appended.push(row); } if (appended.length) sh.getRange(sh.getLastRow() + 1, 1, appended.length, 6).setValues(appended); return { saved: notes.length }; } /* ------------------------------------------------------- daily Drive snapshots */ function folder_() { if (DRIVE_FOLDER_ID) return DriveApp.getFolderById(DRIVE_FOLDER_ID); var it = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME); return it.hasNext() ? it.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME); } function snapName_(day) { return 'DCLM_' + day + '.json.gz'; } function validDay_(day) { if (!/^\d{4}-\d{2}-\d{2}$/.test(String(day || ''))) throw new Error('Bad day "' + day + '" (expected YYYY-MM-DD).'); return day; } /** One base64 part of a gzipped snapshot. Parts are held as temporary files until commit. */ function snapPart_(b) { var day = validDay_(b.day), part = Number(b.part || 0); if (!b.data) throw new Error('snapPart: no data'); var f = folder_(), name = '.part_' + b.uploadId + '_' + part; var old = f.getFilesByName(name); while (old.hasNext()) old.next().setTrashed(true); f.createFile(name, String(b.data), MimeType.PLAIN_TEXT); return { day: day, part: part }; } /** Joins the parts into DCLM_.json.gz, replacing that day's earlier file. */ function snapCommit_(b) { var day = validDay_(b.day), parts = Number(b.parts || 0), f = folder_(), pieces = []; for (var i = 0; i < parts; i++) { var it = f.getFilesByName('.part_' + b.uploadId + '_' + i); if (!it.hasNext()) throw new Error('snapCommit: part ' + i + ' is missing. Re-upload.'); var pf = it.next(); pieces.push(pf.getBlob().getDataAsString()); pf.setTrashed(true); } var bytes = Utilities.base64Decode(pieces.join('')); var old = f.getFilesByName(snapName_(day)); while (old.hasNext()) old.next().setTrashed(true); var file = f.createFile(Utilities.newBlob(bytes, 'application/gzip', snapName_(day))); file.setDescription(JSON.stringify({ day: day, rows: b.rows || 0, fileName: b.fileName || '', uploadedAt: Date.now(), by: who_() })); pruneSnaps_(f); return { day: day, bytes: bytes.length, id: file.getId() }; } function listSnaps_() { var f = folder_(), it = f.getFiles(), out = []; while (it.hasNext()) { var file = it.next(), m = /^DCLM_(\d{4}-\d{2}-\d{2})\.json\.gz$/.exec(file.getName()); if (!m) continue; var d = {}; try { d = JSON.parse(file.getDescription() || '{}'); } catch (x) {} out.push({ day: m[1], rows: d.rows || 0, fileName: d.fileName || '', uploadedAt: d.uploadedAt || file.getLastUpdated().getTime(), by: d.by || '', bytes: file.getSize() }); } out.sort(function (a, b) { return a.day < b.day ? 1 : a.day > b.day ? -1 : 0; }); return out; } /** Returns one base64 part of a day's file ('' = latest day). */ function readSnap_(day, part) { var list = listSnaps_(); if (!list.length) return { day: '', parts: 0 }; var info = day ? list.filter(function (x) { return x.day === day; })[0] : list[0]; if (!info) throw new Error('No saved data for ' + day + '.'); var b64; var it = folder_().getFilesByName(snapName_(info.day)); if (!it.hasNext()) throw new Error('Snapshot file for ' + info.day + ' not found.'); b64 = Utilities.base64Encode(it.next().getBlob().getBytes()); var parts = Math.max(1, Math.ceil(b64.length / SNAP_PART_CHARS)); return { day: info.day, info: info, part: part, parts: parts, data: b64.slice(part * SNAP_PART_CHARS, (part + 1) * SNAP_PART_CHARS) }; } function pruneSnaps_(f) { var cutoff = new Date(Date.now() - SNAP_KEEP_DAYS * 86400000), it = f.getFiles(); while (it.hasNext()) { var file = it.next(), n = file.getName(); if (/^DCLM_.*\.json\.gz$/.test(n) && file.getLastUpdated() < cutoff) file.setTrashed(true); if (/^\.part_/.test(n) && file.getLastUpdated() < new Date(Date.now() - 86400000)) file.setTrashed(true); } } /* --------------------------------------------------------------- manual QA */ /** Run once from the editor to check access and create the helper tabs. */ function setupCheck() { var sh = baseSheet_(); rcaSheet_(); metaSheet_(); Logger.log('Base tab: %s | rows: %s | cols: %s', sh.getName(), sh.getLastRow(), sh.getLastColumn()); Logger.log('Helper tabs ready: %s, %s', RCA_SHEET_NAME, META_SHEET_NAME); var f = folder_(); Logger.log('Snapshot folder: %s (%s) | saved days: %s', f.getName(), f.getUrl(), listSnaps_().length); }
